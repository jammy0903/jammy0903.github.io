**이분탐색** *(binary search)*
templete
```
arr = [1,2,3,4,5]
target = 3
start_idx = 0
end_idx = len(arr)-1  #이유는 0 부터 시작하니까 마지막은 -1을 한 4가 맞지 

# 변수를 함수 내부에서 관리하거나 인자로 전달하는 것이 좋다
def bs(target, start_idx, end_idx):		
				#함수의 역할 : 있으면 true , 없으면 false 
				while (start_idx <= end_idx) :
								mid_idx = (start_idx + end_idx) // 2 #정수결과가 나오는 나누기 
							  if arr[mid_idx] ==target : return (True)
								elif arr[mid_idx] > target: 
												end_idx = mid_idx -1
								else :
								    start_idx = mid_idx + 1
				return False # 루프가 끝날 때까지 못 찾으면 False 반환
				
# 함수 호출 및 결과 출력
# 파이썬의 불리언 값은 True, False로 첫 글자가 대문자
if bs(target, 0, len(arr) - 1): 
    print("yes")
else: 
    print("no")
```
